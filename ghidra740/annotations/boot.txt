# Reset, start-up and the interrupt vectors (parent spec §1.1)
rom      A61C  func      boot_reset                     -              RESET: port set-up, zero page $01-$C9 cleared, settings, ports and UART by DIP switch, self-test sound, RAMCODE copy and bank scan, then the main loop (boot_loop_text, prosody $CA92, boot_loop_play)
rom      A665  label     boot_reentry_cold              -              cold re-entry from another bank (bank_entry_cold): DIP switches and settings again, then boot_reentry_warm
rom      A66B  label     boot_reentry_warm              -              warm re-entry from another bank (bank_entry_warm) and the tail of boot_reset: interrupts on, RAMCODE copy, bank scan, main loop
rom      A68D  func      boot_loop_text                 -              main loop step 1: wait for and take one clause of input (boot_get_clause), then the command scan $ACFA and text conversion $B1E6
rom      A697  func      boot_get_clause                -              wait (isr_idle) until the input ring ($3C read, $3E write) holds data, then copy one clause into $1B00 up to CR, punctuation or the length limit; X = length, also stored in $54
rom      A742  func      boot_set_defaults              -              power-up defaults of the @-command settings in zero page and $1200/$1486/$1827, then $2014 work area cleared ($A7DF) and boot_load_prosody_defaults
rom      A79F  func      boot_selftest_sound            -              self-test (boot_selftest), queue the OK or FAIL sound command string (boot_queue_string) and speak it at once through the main loop steps
rom      A7BE  data      boot_selftest_ok_cmds          string[7]      zero-terminated command string queued when the self-test passes: cancel, then the short buzz sound effect (@T)
rom      A7C5  data      boot_selftest_fail_cmds        string[15]     zero-terminated command string queued when the self-test fails: cancel, then three long @T sound effects
rom      A7D4  func      boot_loop_play                 -              main loop step 3: when an utterance is ready ($6B non-zero) and not aborted ($69 bit 0), play it (continues at boot_play_body)
rom      A7F0  func      boot_load_prosody_defaults     -              copy the 26 default prosody parameters boot_prosody_defaults to pros_params ($168B)
rom      ABF0  func      boot_clear_input_state         -              clear the input/abort flags and reset the input ring pointers $3C/$3E to $2100 and $00/$02 to $3900
rom      AC22  func      boot_init_ports                -              port set-up by interface (DIP bit 0 in $10): P0 function/direction, P1 latch/direction, P2 = $70 (main bank), P3
rom      AC68  func      boot_init_host_lines           -              UART control $D1, P0 function $4C, then the busy line P0.5 and ICR3 from the host handshake input P0.2 (isr_host_line_high/low); serial mode only
rom      AC7E  func      boot_init_baud                 -              UART baud rate from the P3.1/P3.2 jumpers; with X = 1 also speaks the rate (texts at $ACC7/$ACCA/$ACD8 via $A876); clears $53
rom      ACC7  data      boot_baud_text_19200           char[3]        length-prefixed phonetic text (length byte, text, CR) for UART_BAUD $02 (P3.1 = P3.2 = 0; 19200 baud by elimination): silent
rom      ACCA  data      boot_baud_text_300             char[14]       length-prefixed phonetic text (length byte, text, CR) announcing 300 baud (UART_BAUD $BF)
rom      ACD8  data      boot_baud_text_1200            char[16]       length-prefixed phonetic text (length byte, text ending in CR) announcing 1200 baud (UART_BAUD $2F)
rom      ADE0  func      boot_read_dip                  -              A/D off; DIP switches P4 & $0F -> boot_dip ($10)
rom      ADEB  func      boot_init_settings             -              boot_clear_input_state, boot_set_defaults, and the engine version bytes $FFE0/$FFE1 copied to $1819/$181B
rom      B2B3  label     boot_play_body                 -              boot_loop_play body: hold the host, prosody playback set-up $BAF2, isr_speech_start, volume ramp, frame loop $B877, volume off, isr_speech_stop
rom      C605  func      boot_selftest                  -              self-test: boot_selftest_fail ($0F) = 1 when DIP bit 1 is set and input P2.1 is high, else 0
rom      C612  func      boot_queue_string              -              queue the zero-terminated string at ($46) into the input ring ($C3CC) with interrupts off, as if the host had sent it; A and Y preserved
rom      C65B  data      boot_prosody_defaults          byte[26]       power-up values of the 26 prosody parameters pros_params, one per letter A-Z (@R letter command)
rom      FFF4  data      vec_int4_timerx                ptr            INT4 Timer X / CNTR
rom      FFF6  data      vec_int3_step                  ptr            INT3 stepper counters
rom      FFF8  data      vec_int2_timers                ptr            INT2 timers 1-3 (timer 3 = DSP tick)
rom      FFFA  data      vec_int1_serial                ptr            INT1 _INT1 pin and UART receive
rom      FFFC  data      vec_int0                       ptr            INT0 _INT2 pin
rom      FFFE  data      vec_reset                      ptr            RESET
