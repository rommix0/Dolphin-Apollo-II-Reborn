# M3 — Ghidra 740 Module and Project — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A MELPS 740 SLEIGH module validated against `cpu740`, and a Ghidra project of the Apollo 2 ROM
that is rebuilt from committed text annotations. In that project all code is disassembled, every byte is
classified, and every function and table has a name.

**Architecture:**
- **C side:** the emulator records execution coverage, and `cpu740` writes random single-instruction
  vectors.
- **Ghidra side:** a SLEIGH module plus six Java GhidraScripts:
  - `Annot`, the shared parser;
  - `ValidateModule`, `BuildProject`, `CheckProject`, `ExportAnnotations` and `DumpProject`.
- **Install:** `ghidra740\install.bat` copies the module and its scripts into the user's Ghidra
  extension directory.
- **Two rebuild targets:**
  - `rebuild.bat` rebuilds a headless scratch project;
  - the user's open GUI project `apollo_disasm` is rebuilt over MCP.
- **Labelling:** done one subsystem at a time by editing `ghidra740/annotations/*.txt`.

**Tech Stack:**
- C11/MSVC: `build.bat`, /W4 /WX.
- SLEIGH, and Java 21+ GhidraScripts (Ghidra 12.1.3, JDK 27).
- Windows batch.
- The `ghidra-mcp` bridge.

**Spec:** `docs/superpowers/specs/2026-10-01-m3-ghidra740-design.md` (revised 2026-10-02). Its parent is
`docs/superpowers/specs/2026-09-30-software-apollo2-design.md` (rev 5).

```c
#ifndef COVERAGE_H
#define COVERAGE_H
/* Execution coverage of the emulated firmware, for the Ghidra project rebuild (M3 spec §5.1).
   Spaces: rom = main ROM and the default $4000-$7FFF window; low = $4000-$7FFF while P2.7 = 1
   (file 0x0000-0x3FFF); ramcode = $0200-$03FE (the boot copy of $ADFE-$AFFC); ram = other RAM. */
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>

typedef enum { COV_ROM, COV_LOW, COV_RAMCODE, COV_RAM, COV_SPACE_COUNT } cov_space;

typedef struct { uint8_t from_space, to_space; uint16_t from, to; } cov_edge;

typedef struct {
    uint8_t exec[COV_SPACE_COUNT][0x10000 / 8];   /* one bit per executed instruction address */
    cov_edge *edges;                              /* indirect JMP/JSR: unique (from, to) pairs */
    size_t edge_count, edge_cap;
    bool oom;                                     /* an edge could not be stored */
} coverage;

void coverage_init(coverage *c);
void coverage_free(coverage *c);
void coverage_mark(coverage *c, cov_space s, uint16_t pc);
bool coverage_hit(const coverage *c, cov_space s, uint16_t pc);
size_t coverage_count(const coverage *c, cov_space s);
void coverage_add_edge(coverage *c, cov_space from_space, uint16_t from, cov_space to_space, uint16_t to);
const char *coverage_space_name(cov_space s);
/* A header comment, then "x SPACE ADDR" lines in space order and address order, then
   "e FROMSPACE FROM TOSPACE TO" lines sorted the same way. Returns false on an I/O error or lost edge. */
bool coverage_write(const coverage *c, FILE *out);

#endif
```

Create `src/machine/coverage.c`:

```c
#include <stdlib.h>
#include <string.h>
#include "machine/coverage.h"

static const char *const k_space_names[COV_SPACE_COUNT] = { "rom", "low", "ramcode", "ram" };

void coverage_init(coverage *c) { memset(c, 0, sizeof *c); }

void coverage_free(coverage *c)
{
    free(c->edges);
    c->edges = NULL;
    c->edge_count = c->edge_cap = 0;
}

void coverage_mark(coverage *c, cov_space s, uint16_t pc)
{
    c->exec[s][pc >> 3] |= (uint8_t)(1u << (pc & 7));
}

bool coverage_hit(const coverage *c, cov_space s, uint16_t pc)
{
    return ((c->exec[s][pc >> 3] >> (pc & 7)) & 1) != 0;
}

size_t coverage_count(const coverage *c, cov_space s)
{
    size_t n = 0;
    for (unsigned a = 0; a < 0x10000; a++)
        n += coverage_hit(c, s, (uint16_t)a);
    return n;
}

void coverage_add_edge(coverage *c, cov_space from_space, uint16_t from, cov_space to_space, uint16_t to)
{
    for (size_t i = 0; i < c->edge_count; i++) {
        const cov_edge *e = &c->edges[i];
        if (e->from_space == from_space && e->from == from && e->to_space == to_space && e->to == to)
            return;
    }
    if (c->edge_count == c->edge_cap) {
        size_t cap = c->edge_cap ? 2 * c->edge_cap : 64;
        cov_edge *p = realloc(c->edges, cap * sizeof *p);
        if (!p) {
            c->oom = true;
            return;
        }
        c->edges = p;
        c->edge_cap = cap;
    }
    c->edges[c->edge_count++] = (cov_edge){ (uint8_t)from_space, (uint8_t)to_space, from, to };
}

const char *coverage_space_name(cov_space s)
{
    return (unsigned)s < COV_SPACE_COUNT ? k_space_names[s] : "?";
}

static int cmp_edge(const void *pa, const void *pb)
{
    const cov_edge *a = pa, *b = pb;
    if (a->from_space != b->from_space)
        return a->from_space - b->from_space;
    if (a->from != b->from)
        return a->from < b->from ? -1 : 1;
    if (a->to_space != b->to_space)
        return a->to_space - b->to_space;
    return a->to < b->to ? -1 : a->to > b->to;
}

bool coverage_write(const coverage *c, FILE *out)
{
    fprintf(out, "# apollo2 coverage v1: x SPACE ADDR = executed instruction; "
                 "e SPACE FROM SPACE TO = indirect JMP/JSR target\n");
    for (int s = 0; s < COV_SPACE_COUNT; s++)
        for (unsigned a = 0; a < 0x10000; a++)
            if (coverage_hit(c, (cov_space)s, (uint16_t)a))
                fprintf(out, "x %s %04X\n", k_space_names[s], a);
    cov_edge *sorted = NULL;
    if (c->edge_count) {
        sorted = malloc(c->edge_count * sizeof *sorted);
        if (!sorted)
            return false;
        memcpy(sorted, c->edges, c->edge_count * sizeof *sorted);
        qsort(sorted, c->edge_count, sizeof *sorted, cmp_edge);
    }
    for (size_t i = 0; i < c->edge_count; i++)
        fprintf(out, "e %s %04X %s %04X\n", k_space_names[sorted[i].from_space], sorted[i].from,
                k_space_names[sorted[i].to_space], sorted[i].to);
    free(sorted);
    return !ferror(out) && !c->oom;
}
```

```c
/* Dev tool (M3 spec §5.1): runs the firmware through boot and every golden-corpus text and writes the
   union of the executed instruction addresses and the indirect JMP/JSR targets, for ghidra740. */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include "engine/rom_engine.h"
#include "machine/coverage.h"
#include "../../tests/corpus_list.h"

static machine g_m;
static coverage g_cov;
static uint8_t g_rom[MACHINE_ROM_SIZE];
static const char *const k_corpus[] = { CORPUS_NAMES };

static int usage(void)
{
    fprintf(stderr, "usage: apollo2_coverage [--rom PATH] -o FILE [TEXTFILE...]\n"
                    "  runs boot, every tests/corpus/<name>.txt of the golden corpus, then each TEXTFILE\n"
                    "  --rom PATH  ROM image (default: $APOLLO2_ROM or UK_V2.01_4.04_ROM.BIN)\n");
    return 1;
}

static bool load_rom(const char *path)
{
    FILE *f = fopen(path, "rb");
    if (!f)
        return false;
    bool ok = fread(g_rom, 1, sizeof g_rom, f) == sizeof g_rom && fgetc(f) == EOF;
    fclose(f);
    return ok;
}

/* Whole file, binary, trailing CR/LF removed (as the golden test does). NULL if unreadable. */
static char *read_text(const char *path)
{
    FILE *f = fopen(path, "rb");
    if (!f)
        return NULL;
    fseek(f, 0, SEEK_END);
    long size = ftell(f);
    fseek(f, 0, SEEK_SET);
    char *s = size >= 0 ? malloc((size_t)size + 1) : NULL;
    size_t n = s ? fread(s, 1, (size_t)size, f) : 0;
    fclose(f);
    if (!s)
        return NULL;
    while (n > 0 && (s[n - 1] == '\n' || s[n - 1] == '\r'))
        n--;
    s[n] = '\0';
    return s;
}

static bool run(const char *label, const char *text, bool boot)
{
```

  - the vector line format below, which `ValidateModule.java` (Task 5) parses:

```
V pc=PPPP op=XX[,XX[,XX]] a=XX x=XX y=XX s=XX p=XX r=[AAAA:VV[,AAAA:VV...]] | A=XX X=XX Y=XX S=XX P=XX PC=PPPP w=[AAAA:VV,...] | TEXT
D pc=PPPP op=XX[,XX[,XX]] | TEXT
```

- `V` lines are executed vectors. `r` lists every byte the instruction read that is not one of its
  own bytes, with the value it read. `w` lists every byte it wrote, with the last value written.
- `D` lines are decode-only. They cover opcodes that `cpu740` faults on, or that change no
  architectural state (WIT): ILL, WIT, BRK, SED, SET, STP, MUL and DIV.
- The `p` value always has D = 0, T = 0 and B = 1. The `s` value lies in `$10–$EF`, so pushes and
  pulls never wrap the stack page.

```c
   One line per vector:

   p always has D = 0, T = 0, B = 1; s is in $10-$EF, so the stack page never wraps.
   per_opcode V vectors are written for each executable opcode, one D line for each decode-only one. */
```

Create `src/cpu740/vec740.c`:

```c
#include <stdbool.h>
#include <string.h>
#include "cpu740/cpu740.h"
#include "cpu740/optable.h"
#include "cpu740/vec740.h"

#define VEC_MAX_ACCESS 16

/* Flat memory whose unknown bytes get random values on first read, which are then logged. */
typedef struct {
    uint8_t mem[0x10000];
    uint8_t known[0x10000 / 8];
    uint32_t rng;
    uint16_t rd_addr[VEC_MAX_ACCESS], wr_addr[VEC_MAX_ACCESS];
    uint8_t rd_val[VEC_MAX_ACCESS], wr_val[VEC_MAX_ACCESS];
    int nrd, nwr;
    bool overflow;
} vecmem;

static vecmem g_vm;

static uint32_t rnd(uint32_t *s)
{
    uint32_t x = *s;
    x ^= x << 13;
    x ^= x >> 17;
    x ^= x << 5;
    return *s = x;
}

static bool is_known(const vecmem *v, uint16_t a) { return ((v->known[a >> 3] >> (a & 7)) & 1) != 0; }
static void set_known(vecmem *v, uint16_t a) { v->known[a >> 3] |= (uint8_t)(1u << (a & 7)); }

static uint8_t vm_read(void *ctx, uint16_t a)
{
    vecmem *v = ctx;
    if (!is_known(v, a)) {
        v->mem[a] = (uint8_t)rnd(&v->rng);
        set_known(v, a);
        if (v->nrd < VEC_MAX_ACCESS) {
            v->rd_addr[v->nrd] = a;
            v->rd_val[v->nrd++] = v->mem[a];
        } else {
            v->overflow = true;
        }
    }
    return v->mem[a];
}

static void vm_write(void *ctx, uint16_t a, uint8_t val)
{
    vecmem *v = ctx;
    set_known(v, a);
    v->mem[a] = val;
    for (int i = 0; i < v->nwr; i++) {
        if (v->wr_addr[i] == a) {
            v->wr_val[i] = val;
            return;
        }
    }
    if (v->nwr < VEC_MAX_ACCESS) {
        v->wr_addr[v->nwr] = a;
        v->wr_val[v->nwr++] = val;
    } else {
        v->overflow = true;
    }
}

static uint8_t vm_peek(void *ctx, uint16_t a) { return ((const vecmem *)ctx)->mem[a]; }

/* Operand bytes: mostly random, with $FF and $00 often enough to reach the zero-page and page wraps. */
static uint8_t operand_byte(uint32_t *rng)
{
    uint32_t r = rnd(rng);
    switch (r & 7) {
    case 0:  return 0xFF;
    case 1:  return 0x00;
    default: return (uint8_t)(r >> 8);
    }
}

static bool decode_only(int mnem)
{
    switch (mnem) {
    case M_ILL: case M_WIT: case M_BRK: case M_SED: case M_SET: case M_STP: case M_MUL: case M_DIV:
        return true;
    default:
        return false;
    }
}

static void put_list(char *out, size_t n, const uint16_t *addr, const uint8_t *val, int count)
{
    size_t pos = 0;
    out[0] = '\0';
    for (int i = 0; i < count && pos + 9 < n; i++)
        pos += (size_t)snprintf(out + pos, n - pos, "%s%04X:%02X", i ? "," : "", addr[i], val[i]);
}

int vec740_write(FILE *out, unsigned per_opcode, uint32_t seed)
{
    const cpu740_opinfo *table = cpu740_optable();
    const cpu740_bus bus = { &g_vm, vm_read, vm_write };
    uint32_t rng = seed ? seed : 1;
    int lines = 0;

    for (int op = 0; op < 256; op++) {
        int mnem = table[op].mnem;
        int len = mnem == M_ILL ? 1 : cpu740_mode_length(table[op].mode);
        unsigned want = decode_only(mnem) ? 1 : per_opcode, done = 0, tries = 0;
        while (done < want && tries++ < 20 * want + 20) {
            uint8_t bytes[3] = { 0 };
            char ops[12], text[48], rd[VEC_MAX_ACCESS * 8 + 1], wr[VEC_MAX_ACCESS * 8 + 1];
            uint16_t pc = (uint16_t)(0x4000 + rnd(&rng) % 0x3FF0);
            memset(&g_vm, 0, sizeof g_vm);
            bytes[0] = (uint8_t)op;
            for (int i = 1; i < len; i++)
                bytes[i] = operand_byte(&rng);
            for (int i = 0; i < len; i++) {
                g_vm.mem[(uint16_t)(pc + i)] = bytes[i];
                set_known(&g_vm, (uint16_t)(pc + i));
            }
            if (len == 1)
                snprintf(ops, sizeof ops, "%02X", bytes[0]);
            else if (len == 2)
                snprintf(ops, sizeof ops, "%02X,%02X", bytes[0], bytes[1]);
            else
                snprintf(ops, sizeof ops, "%02X,%02X,%02X", bytes[0], bytes[1], bytes[2]);
            cpu740_disasm(pc, vm_peek, &g_vm, text, sizeof text);
            if (decode_only(mnem)) {
                fprintf(out, "D pc=%04X op=%s | %s\n", pc, ops, text);
                lines++;
                done++;
                continue;
            }

            cpu740 c;
            memset(&c, 0, sizeof c);
            c.a = (uint8_t)rnd(&rng);
            c.x = (uint8_t)rnd(&rng);
            c.y = (uint8_t)rnd(&rng);
            c.s = (uint8_t)(0x10 + rnd(&rng) % 0xE0);
            c.p = (uint8_t)((rnd(&rng) & (uint32_t)~(CPU740_D | CPU740_T)) | CPU740_B);
            c.pc = pc;
            const cpu740 before = c;
            g_vm.rng = rnd(&rng) | 1;
            if (cpu740_step(&c, &bus) == 0 || c.waiting || g_vm.overflow)
                continue;   /* e.g. PLP/RTI pulling T = 1: try another state */
            put_list(rd, sizeof rd, g_vm.rd_addr, g_vm.rd_val, g_vm.nrd);
            put_list(wr, sizeof wr, g_vm.wr_addr, g_vm.wr_val, g_vm.nwr);
            fprintf(out, "V pc=%04X op=%s a=%02X x=%02X y=%02X s=%02X p=%02X r=%s | "
                         "A=%02X X=%02X Y=%02X S=%02X P=%02X PC=%04X w=%s | %s\n",
                    pc, ops, before.a, before.x, before.y, before.s, before.p, rd,
                    c.a, c.x, c.y, c.s, c.p, c.pc, wr, text);
            lines++;
            done++;
        }
        if (done < want)
            fprintf(out, "# opcode %02X: only %u of %u vectors (cpu740 faulted on the other tries)\n", op, done, want);
    }
    return ferror(out) ? -1 : lines;
}
```

```c
int main(int argc, char **argv)
{
    const char *out_path = NULL, *blank_path = NULL;
    unsigned n = 64;
    uint32_t seed = 1;

            out_path = argv[++i];

        fprintf(stderr, "cannot write %s\n", out_path);
        return 1;

        }
    }
    printf("%d vector lines written to %s\n", lines, out_path);

}
```

```
# Derived from Ghidra's 6502.slaspec (Apache License 2.0). The semantics follow the project's oracle,
# src/cpu740/cpu740.c, and ghidra740\validate.bat checks them against it.
# Not modelled:
#   - decimal arithmetic (D is only stored);
#   - T-flag addressing (T is only stored; the firmware never sets it);
#   - stack-page wrap (SP = $01xx, and only S, its low byte, is meant to move: keep S clear of $00/$FF).

define alignment=1;

define space RAM      type=ram_space      size=2  default;
define space register type=register_space size=1;

define register offset=0x00 size=1 [ A X Y ];
define register offset=0x20 size=2 [ PC SP ];
define register offset=0x20 size=1 [ PCL PCH S SH ];
define register offset=0x30 size=1 [ N V T B D I Z C ];

   op   = (0,7)
   aaa  = (5,7)
   bitn = (5,7) dec
   bbb  = (2,4)
   cc   = (0,1)

   imm8 = (0,7)
   rel  = (0,7) signed

   zp2  = (0,7)
   rel2 = (0,7) signed

define pcodeop stop;

macro resultFlags(value) {
   Z = (value == 0);
   N = (value s< 0);
}

macro addWithCarry(operand) {
   local sum:2 = zext(A) + zext(operand) + zext(C);
   local r:1 = sum:1;
   V = ((~(A ^ operand) & (A ^ r)) & 0x80) != 0;
   C = sum > 0xff;
   A = r;
   resultFlags(A);
}

macro push8(v) {
   *:1 SP = v;
   SP = SP - 1;
}

macro pull8(v) {
   SP = SP + 1;
   v = *:1 SP;
}

# cpu740 pushes the high byte at S, the low byte at S-1, and leaves S-2.
macro push16(v) {
   SP = SP - 1;
   *:2 SP = v;
   SP = SP - 1;
}

macro pull16(v) {
   SP = SP + 1;
   v = *:2 SP;
   SP = SP + 1;
}

macro packP(r) {
   r = (N << 7) | (V << 6) | (T << 5) | (B << 4) | (D << 3) | (I << 2) | (Z << 1) | C;
}

# PLP/RTI: B always reads back as 1 (cpu740 ORs it in).
macro unpackP(r) {
   N = (r >> 7) & 1;
   V = (r >> 6) & 1;
   T = (r >> 5) & 1;
   B = 1;
   D = (r >> 3) & 1;
   I = (r >> 2) & 1;
   Z = (r >> 1) & 1;
   C = r & 1;
}
```

```java
// build/ghidra740/validate.txt out. Run headless on a blank program (ghidra740\validate.bat).
//@category ghidra740
import ghidra.app.emulator.EmulatorHelper;
import ghidra.app.script.GhidraScript;
import ghidra.app.util.PseudoDisassembler;
import ghidra.app.util.PseudoInstruction;
import ghidra.program.model.address.*;
import java.math.BigInteger;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.*;
import java.util.regex.*;

public class ValidateModule extends GhidraScript {
    private static final String[] FLAG_BITS = { "C", "Z", "I", "D", "B", "T", "V", "N" };   // P bit 0..7
    private static final Pattern NUM = Pattern.compile("0X([0-9A-F]+)|\\$([0-9A-F]+)");
    private final List<String> mismatches = new ArrayList<>();
    private int executed, decodeOnly;

    @Override
    public void run() throws Exception {
        Path root = Annot.repoRoot(this);
        Path in = root.resolve("build/ghidra740/vectors.txt");
        Path out = root.resolve("build/ghidra740/validate.txt");
        PseudoDisassembler pd = new PseudoDisassembler(currentProgram);
        AddressSpace ram = currentProgram.getAddressFactory().getDefaultAddressSpace();
        for (String line : Files.readAllLines(in, StandardCharsets.US_ASCII)) {
            if (line.isBlank() || line.startsWith("#"))
                continue;
            monitor.checkCancelled();
            check(line, pd, ram);
        }
        StringBuilder sb = new StringBuilder();
        for (String m : mismatches)
            sb.append(m).append('\n');
        sb.append(String.format("executed %d, decode-only %d, mismatches %d%n", executed, decodeOnly, mismatches.size()));
        sb.append(mismatches.isEmpty() ? "RESULT PASS\n" : "RESULT FAIL\n");
        Files.writeString(out, sb.toString(), StandardCharsets.US_ASCII);
        println("ValidateModule: " + (mismatches.isEmpty() ? "PASS" : mismatches.size() + " mismatches") + " -> " + out);
    }

    /** Case, spacing and the "\" of JSR \$FFxx removed; every hex number ($12, 0x12) rewritten in decimal. */
    static String canon(String s) {
        String u = s.toUpperCase(Locale.ROOT).replace("\\", "");
        Matcher m = NUM.matcher(u);
        StringBuilder sb = new StringBuilder();
        while (m.find()) {
            String h = m.group(1) != null ? m.group(1) : m.group(2);
            m.appendReplacement(sb, Long.toString(Long.parseLong(h, 16)));
        }
        m.appendTail(sb);
        return sb.toString().replaceAll("\\s+", "");
    }

    private static Map<String, String> fields(String part) {
        Map<String, String> f = new HashMap<>();
        for (String tok : part.trim().split("\\s+")) {
            int eq = tok.indexOf('=');
            if (eq > 0)
                f.put(tok.substring(0, eq), tok.substring(eq + 1));
        }
        return f;
    }

    private static int hex(String s) { return Integer.parseInt(s, 16); }

    private static byte[] bytes(String list) {
        String[] p = list.split(",");
        byte[] b = new byte[p.length];
        for (int i = 0; i < p.length; i++)
            b[i] = (byte) hex(p[i]);
        return b;
    }

    /** "AAAA:VV,AAAA:VV" -> ordered map; empty string -> empty map. */
    private static Map<Integer, Integer> pairs(String list) {
        Map<Integer, Integer> m = new LinkedHashMap<>();
        if (list == null || list.isEmpty())
            return m;
        for (String p : list.split(",")) {
            String[] kv = p.split(":");
            m.put(hex(kv[0]), hex(kv[1]));
        }
        return m;
    }

    private void check(String line, PseudoDisassembler pd, AddressSpace ram) throws Exception {
        String[] parts = line.split(" \\| ");
        Map<String, String> pre = fields(parts[0]);
        int pc = hex(pre.get("pc"));
        byte[] code = bytes(pre.get("op"));
        String want = parts[parts.length - 1].trim();
        Address at = ram.getAddress(pc);
        String got;
        try {
            PseudoInstruction ins = pd.disassemble(at, code);
            got = ins == null ? null : ins.toString();
        } catch (Exception e) {
            got = null;
        }
        boolean illegal = want.startsWith("DB ");
        if (illegal ? got != null : got == null || !canon(got).equals(canon(want)))
            mismatches.add(String.format("DISASM %s: ghidra \"%s\", cpu740 \"%s\"", pre.get("op"), got, want));
        if (line.startsWith("D ")) {
            decodeOnly++;
            return;
        }
        executed++;
        emulate(line, pre, fields(parts[1]), code, at, pc, ram);
    }

    private void emulate(String line, Map<String, String> pre, Map<String, String> post, byte[] code,
                         Address at, int pc, AddressSpace ram) throws Exception {
        EmulatorHelper emu = new EmulatorHelper(currentProgram);
        try {
            for (Map.Entry<Integer, Integer> m : pairs(pre.get("r")).entrySet())
                emu.writeMemoryValue(ram.getAddress(m.getKey()), 1, m.getValue());
            emu.writeMemory(at, code);
            emu.writeRegister("A", hex(pre.get("a")));
            emu.writeRegister("X", hex(pre.get("x")));
            emu.writeRegister("Y", hex(pre.get("y")));
            emu.writeRegister("SP", 0x100 | hex(pre.get("s")));
            int p = hex(pre.get("p"));
            for (int b = 0; b < 8; b++)
                emu.writeRegister(FLAG_BITS[b], (p >> b) & 1);
            emu.writeRegister(emu.getPCRegister(), pc);
            emu.enableMemoryWriteTracking(true);
            if (!emu.step(monitor)) {
                mismatches.add("EMU " + pre.get("op") + ": step failed: " + emu.getLastError() + " | " + line);
                return;
            }
            List<String> diffs = new ArrayList<>();
            cmp(diffs, "A", emu.readRegister("A").intValue(), hex(post.get("A")));
            cmp(diffs, "X", emu.readRegister("X").intValue(), hex(post.get("X")));
            cmp(diffs, "Y", emu.readRegister("Y").intValue(), hex(post.get("Y")));
            int sp = emu.readRegister("SP").intValue();
            cmp(diffs, "S", sp & 0xFF, hex(post.get("S")));
            cmp(diffs, "SH", sp >> 8, 1);
            int gp = 0;
            for (int b = 0; b < 8; b++)
                gp |= (emu.readRegister(FLAG_BITS[b]).intValue() & 1) << b;
            cmp(diffs, "P", gp, hex(post.get("P")));
            cmp(diffs, "PC", emu.readRegister(emu.getPCRegister()).intValue(), hex(post.get("PC")));
            Map<Integer, Integer> wantW = pairs(post.get("w"));
            Set<Integer> gotW = new TreeSet<>();
            for (AddressRange r : emu.getTrackedMemoryWriteSet())
                for (Address a = r.getMinAddress(); a != null && a.compareTo(r.getMaxAddress()) <= 0; a = a.next())
                    if (a.getAddressSpace().equals(ram))
                        gotW.add((int) a.getOffset());
            if (!gotW.equals(wantW.keySet()))
                diffs.add("writes at " + hexSet(gotW) + " want " + hexSet(wantW.keySet()));
            for (Map.Entry<Integer, Integer> w : wantW.entrySet())
                cmp(diffs, String.format("mem[%04X]", w.getKey()),
                    emu.readMemoryByte(ram.getAddress(w.getKey())) & 0xFF, w.getValue());
            if (!diffs.isEmpty())
                mismatches.add("EMU " + pre.get("op") + ": " + String.join(", ", diffs) + " | " + line);
        } finally {
            emu.dispose();
        }
    }

    private static void cmp(List<String> diffs, String what, int got, int want) {
        if (got != want)
            diffs.add(String.format("%s got %X want %X", what, got, want));
    }

    private static String hexSet(Collection<Integer> s) {
        StringBuilder sb = new StringBuilder("{");
        for (int v : s)
            sb.append(sb.length() > 1 ? "," : "").append(String.format("%04X", v));
        return sb.append('}').toString();
    }
}
```

- Create: `ghidra740/module/MELPS740/ghidra_scripts/BuildProject.java`, `ghidra740/rebuild.bat` (first version), and in `ghidra740/annotations/`: `types.txt`, `sfr.txt`, `boot.txt`

**Interfaces:**
- Consumes:
  - Task 2: `ghidra740/coverage/corpus.cov`;
  - Task 5: `Annot.repoRoot`.
- Produces, used by Tasks 7–9 and 11–19:
  - `Annot.load(Path) → Annot.Annotations` (with `recs`, `structs` and `headers`);
  - `Annot.write(Annotations, Path)`;
  - `Annot.loadCoverage(Path) → Annot.Cov` (with `exec` and `edges`);
  - `Annot.addr(Program, String space, int addr) → Address`;
  - `Annot.spaceName(Address) → "ram"|"rom"|"low"|"ramcode"`;
  - `Annot.parseType(String) → TypeRef`, `Annot.BM_CATEGORY = "ghidra740"`, `Annot.ORDER`;
  - memory blocks `ZP SFR RAM ROM_LO ROM` plus overlays `LOW` and `RAMCODE`;
  - `build/ghidra740/build.txt` (`ERROR …` lines and a `SUMMARY …` line);
  - bookmarks of type NOTE, category `ghidra740`, with the description `entry[: comment]`, `jumptable`
    or `jumptable-m1`.

**Annotation format.** This makes the spec's §5.3 precise. The spec rev in Task 20 records it.
- **Records:** `SPACE ADDR KIND NAME TYPE [COMMENT]`.
  - Fields are separated by whitespace, and the comment is the rest of the line.
  - `ADDR` is 4 hex digits. `NAME` and `TYPE` are `-` when they don't apply.
  - Only a first line starting with `#` survives an export, as the file's header. Other comment and
    blank lines are allowed but get dropped by `--format`.
- **Spaces:** `ram` (< `$4000`, default space), `rom` (≥ `$4000`, default space), `low` (overlay,
  `$4000–$7FFF`) and `ramcode` (overlay, `$0200–$03FE`).
- **Kinds:**
  - `func`: the comment becomes the plate comment;
  - `label` and `data`: the comment becomes the EOL comment;
  - `jumptable`: type `ptr[n]`, or `ptrm1[n]` for tables holding target−1 (RTS dispatch);
  - `entry`: a disassembly seed. Its name is `-`, and its comment is kept in a bookmark;
  - `comment`: an EOL comment, with name `-`.
- **Types:** `byte word ptr ptrm1 char string`, `T[n]`, and struct names from `types.txt`.
  - `string` needs `[n]`.
  - Struct fields use only the primitive types, and arrays of them, but not `string` (use `char[n]`).
- **`types.txt`:** `struct NAME SIZE [COMMENT]` and `field STRUCT OFFSET NAME TYPE [COMMENT]`.
- **Rules:**
  - at most one named record (`func`/`label`/`data`/`jumptable`) per space and address;
  - at most one EOL comment per space and address;
  - unique names;
  - no overlapping data.

```java
// The ghidra740 annotation files (M3 spec §5.3) and coverage file (§5.1). Not a script: shared by
// BuildProject, CheckProject, DumpProject, ExportAnnotations and ValidateModule.
import ghidra.app.script.GhidraScript;
import ghidra.program.model.address.*;

import java.util.stream.*;

public class Annot {
    public static final List<String> SPACES = List.of("ram", "rom", "low", "ramcode");
    public static final List<String> KINDS = List.of("func", "label", "data", "jumptable", "entry", "comment");
    public static final List<String> PRIMITIVES = List.of("byte", "word", "ptr", "ptrm1", "char", "string");
    public static final String TYPES_FILE = "types.txt";
    public static final String NEW_FILE = "new.txt";
    public static final String BM_CATEGORY = "ghidra740";

    public static final class Rec {
        public final String file, space, kind, name, type, comment;
        public final int line, addr;

        public Rec(String file, int line, String space, int addr, String kind, String name, String type, String comment) {
            this.file = file; this.line = line; this.space = space; this.addr = addr;
            this.kind = kind; this.name = name; this.type = type; this.comment = comment == null ? "" : comment;
        }
        public boolean named() { return !name.equals("-"); }
        public String where() { return file + ":" + line; }
        public String format() {
            return String.format("%-8s %04X  %-9s %-30s %-14s %s", space, addr, kind, name, type, comment).stripTrailing();
        }
    }

    public static final class Field {
        public final int offset; public final String name, type, comment;
        public Field(int offset, String name, String type, String comment) {
            this.offset = offset; this.name = name; this.type = type; this.comment = comment == null ? "" : comment;
        }
    }

    public static final class Struct {
        public final String name, comment; public final int size; public final List<Field> fields = new ArrayList<>();
        public Struct(String name, int size, String comment) { this.name = name; this.size = size; this.comment = comment == null ? "" : comment; }
    }

    public static final class Annotations {
        public final List<Rec> recs = new ArrayList<>();
        public final Map<String, Struct> structs = new TreeMap<>();
        public final Map<String, String> headers = new TreeMap<>();   // file name -> first-line comment
    }

    public static final class TypeRef {
        public final String base; public final int count; public final boolean array;
        TypeRef(String base, int count, boolean array) { this.base = base; this.count = count; this.array = array; }
    }

    public static final class Cov {
        public final List<String[]> exec = new ArrayList<>();    // {space, hexaddr}
        public final List<String[]> edges = new ArrayList<>();   // {fromspace, hexfrom, tospace, hexto}
    }

    public static final Comparator<Rec> ORDER = Comparator
        .comparingInt((Rec r) -> SPACES.indexOf(r.space))
        .thenComparingInt(r -> r.addr)
        .thenComparingInt(r -> KINDS.indexOf(r.kind));

    private static final Pattern REC = Pattern.compile("^(\\S+)\\s+([0-9A-Fa-f]{4})\\s+(\\S+)\\s+(\\S+)\\s+(\\S+)(?:\\s+(.*?))?\\s*$");
    private static final Pattern STRUCT = Pattern.compile("^struct\\s+(\\S+)\\s+(\\d+)(?:\\s+(.*?))?\\s*$");
    private static final Pattern FIELD = Pattern.compile("^field\\s+(\\S+)\\s+(\\d+)\\s+(\\S+)\\s+(\\S+)(?:\\s+(.*?))?\\s*$");
    private static final Pattern TYPE = Pattern.compile("^([A-Za-z_][A-Za-z0-9_]*)(?:\\[(\\d+)\\])?$");
    private static final Pattern NAME = Pattern.compile("^[A-Za-z_][A-Za-z0-9_]*$");

    /** The repo root that ghidra740\install.bat wrote to repo.txt beside ghidra_scripts/. */
    public static Path repoRoot(GhidraScript script) throws IOException {
        Path ext = Paths.get(script.getSourceFile().getParentFile().getParentFile().getAbsolutePath());
        return Paths.get(Files.readString(ext.resolve("repo.txt"), StandardCharsets.UTF_8).trim());
    }

    public static TypeRef parseType(String t) {
        Matcher m = TYPE.matcher(t);
        if (!m.matches())
            return null;
        return m.group(2) == null ? new TypeRef(m.group(1), 1, false) : new TypeRef(m.group(1), Integer.parseInt(m.group(2)), true);
    }

    /** Size in bytes, or -1 for an unknown base type. */
    public static int sizeOf(TypeRef t, Map<String, Struct> structs) {
        int e;
        switch (t.base) {
            case "byte": case "char": case "string": e = 1; break;
            case "word": case "ptr": case "ptrm1": e = 2; break;
            default: e = structs.containsKey(t.base) ? structs.get(t.base).size : -1;
        }
        return e < 0 ? -1 : e * t.count;
    }

    public static boolean inRange(String space, int addr) {
        switch (space) {
            case "ram": return addr < 0x4000;
            case "rom": return addr >= 0x4000;
            case "low": return addr >= 0x4000 && addr < 0x8000;
            case "ramcode": return addr >= 0x0200 && addr <= 0x03FE;
            default: return false;
        }
    }

    public static Annotations load(Path dir) throws IOException {
        Annotations a = new Annotations();
        List<String> errors = new ArrayList<>();
        Path types = dir.resolve(TYPES_FILE);
        if (Files.exists(types))
            loadTypes(types, a, errors);
        List<Path> files;
        try (Stream<Path> s = Files.list(dir)) {
            files = s.filter(p -> p.getFileName().toString().endsWith(".txt") && !p.getFileName().toString().equals(TYPES_FILE))
                     .sorted().collect(Collectors.toList());
        }
        for (Path p : files)
            loadRecords(p, a, errors);
        validate(a, errors);
        if (!errors.isEmpty())
            throw new IllegalArgumentException("annotation errors:\n  " + String.join("\n  ", errors));
        return a;
    }

    private static void loadTypes(Path p, Annotations a, List<String> errors) throws IOException {
        List<String> lines = Files.readAllLines(p, StandardCharsets.UTF_8);
        for (int i = 0; i < lines.size(); i++) {
            String line = lines.get(i).strip();
            if (i == 0 && line.startsWith("#")) { a.headers.put(TYPES_FILE, line); continue; }
            if (line.isEmpty() || line.startsWith("#")) continue;
            Matcher s = STRUCT.matcher(line), f = FIELD.matcher(line);
            if (s.matches()) {
                a.structs.put(s.group(1), new Struct(s.group(1), Integer.parseInt(s.group(2)), s.group(3)));
            } else if (f.matches()) {
                Struct st = a.structs.get(f.group(1));
                if (st == null) errors.add(TYPES_FILE + ":" + (i + 1) + ": field of undeclared struct " + f.group(1));
                else st.fields.add(new Field(Integer.parseInt(f.group(2)), f.group(3), f.group(4), f.group(5)));
            } else {
                errors.add(TYPES_FILE + ":" + (i + 1) + ": malformed line");
            }
        }
    }

    private static void loadRecords(Path p, Annotations a, List<String> errors) throws IOException {
        String file = p.getFileName().toString();
        List<String> lines = Files.readAllLines(p, StandardCharsets.UTF_8);
        for (int i = 0; i < lines.size(); i++) {
            String line = lines.get(i).strip();
            if (i == 0 && line.startsWith("#")) { a.headers.put(file, line); continue; }
            if (line.isEmpty() || line.startsWith("#")) continue;
            Matcher m = REC.matcher(line);
            if (!m.matches()) { errors.add(file + ":" + (i + 1) + ": malformed record"); continue; }
            a.recs.add(new Rec(file, i + 1, m.group(1), Integer.parseInt(m.group(2), 16), m.group(3), m.group(4), m.group(5), m.group(6)));
        }
    }

    private static void validate(Annotations a, List<String> errors) {
        for (Struct s : a.structs.values()) {
            for (Field f : s.fields) {
                TypeRef t = parseType(f.type);
                if (t == null || !PRIMITIVES.contains(t.base) || t.base.equals("string"))
                    errors.add(TYPES_FILE + ": " + s.name + "." + f.name + ": field type must be a primitive (not string): " + f.type);
                else if (f.offset + sizeOf(t, a.structs) > s.size)
                    errors.add(TYPES_FILE + ": " + s.name + "." + f.name + " runs past the struct size " + s.size);
            }
        }
        Set<String> names = new HashSet<>(), namedAt = new HashSet<>(), eolAt = new HashSet<>(), kindAt = new HashSet<>();
        List<int[]> spans = new ArrayList<>();   // {spaceIndex, start, end, recIndex}
        for (int i = 0; i < a.recs.size(); i++) {
            Rec r = a.recs.get(i);
            String at = r.space + ":" + String.format("%04X", r.addr);
            if (!SPACES.contains(r.space)) { errors.add(r.where() + ": unknown space " + r.space); continue; }
            if (!KINDS.contains(r.kind)) { errors.add(r.where() + ": unknown kind " + r.kind); continue; }
            if (!inRange(r.space, r.addr)) errors.add(r.where() + ": address " + at + " outside its space");
            if (!kindAt.add(at + ":" + r.kind)) errors.add(r.where() + ": duplicate " + r.kind + " at " + at);
            boolean wantsName = !r.kind.equals("entry") && !r.kind.equals("comment");
            if (wantsName != r.named()) errors.add(r.where() + ": " + r.kind + (wantsName ? " needs a name" : " takes no name (-)"));
            if (r.named()) {
                if (!NAME.matcher(r.name).matches()) errors.add(r.where() + ": bad name " + r.name);
                if (!names.add(r.name)) errors.add(r.where() + ": duplicate name " + r.name);
                if (!namedAt.add(at)) errors.add(r.where() + ": a second named record at " + at);
            }
            if (!r.kind.equals("func") && !r.kind.equals("entry") && !r.comment.isEmpty() && !eolAt.add(at))
                errors.add(r.where() + ": a second EOL comment at " + at);
            boolean typed = r.kind.equals("data") || r.kind.equals("jumptable");
            if (!typed) {
                if (!r.type.equals("-")) errors.add(r.where() + ": " + r.kind + " takes no type (-)");
                continue;
            }
            TypeRef t = parseType(r.type);
            int size = t == null ? -1 : sizeOf(t, a.structs);
            if (size <= 0) { errors.add(r.where() + ": bad type " + r.type); continue; }
            if (t.base.equals("string") && !t.array) errors.add(r.where() + ": string needs a length, string[n]");
            if (r.kind.equals("jumptable") && (!t.array || !(t.base.equals("ptr") || t.base.equals("ptrm1"))))
                errors.add(r.where() + ": jumptable type must be ptr[n] or ptrm1[n]");
            if (r.kind.equals("data") && t.base.equals("ptrm1"))
                errors.add(r.where() + ": ptrm1 is only for jumptable records");
            int sp = r.space.equals("rom") ? 0 : SPACES.indexOf(r.space);   // ram and rom share the default space
            spans.add(new int[] { sp, r.addr, r.addr + size - 1, i });
        }
        spans.sort(Comparator.<int[]>comparingInt(s -> s[0]).thenComparingInt(s -> s[1]));
        for (int i = 1; i < spans.size(); i++) {
            int[] p = spans.get(i - 1), q = spans.get(i);
            if (p[0] == q[0] && q[1] <= p[2])
                errors.add(a.recs.get(q[3]).where() + ": overlaps " + a.recs.get(p[3]).name + " (" + a.recs.get(p[3]).where() + ")");
        }
    }

    /** Writes every file in canonical form: header, then records in ORDER (types.txt: structs by name). */
    public static void write(Annotations a, Path dir) throws IOException {
        Files.createDirectories(dir);
        Map<String, List<Rec>> byFile = new TreeMap<>();
        for (Rec r : a.recs)
            byFile.computeIfAbsent(r.file, k -> new ArrayList<>()).add(r);
        Set<String> files = new TreeSet<>(byFile.keySet());
        files.addAll(a.headers.keySet());
        files.remove(TYPES_FILE);
        for (String f : files) {
            StringBuilder sb = new StringBuilder();
            if (a.headers.containsKey(f)) sb.append(a.headers.get(f)).append('\n');
            List<Rec> rs = byFile.getOrDefault(f, new ArrayList<>());
            rs.sort(ORDER);
            for (Rec r : rs) sb.append(r.format()).append('\n');
            Files.writeString(dir.resolve(f), sb.toString(), StandardCharsets.UTF_8);
        }
        if (a.structs.isEmpty() && !a.headers.containsKey(TYPES_FILE))
            return;
        StringBuilder sb = new StringBuilder();
        if (a.headers.containsKey(TYPES_FILE)) sb.append(a.headers.get(TYPES_FILE)).append('\n');
        for (Struct s : a.structs.values()) {
            sb.append(String.format("%-6s %-24s %5d  %s", "struct", s.name, s.size, s.comment).stripTrailing()).append('\n');
            List<Field> fs = new ArrayList<>(s.fields);
            fs.sort(Comparator.comparingInt(f -> f.offset));
            for (Field f : fs)
                sb.append(String.format("%-6s %-24s %5d  %-24s %-10s %s", "field", s.name, f.offset, f.name, f.type, f.comment).stripTrailing()).append('\n');
        }
        Files.writeString(dir.resolve(TYPES_FILE), sb.toString(), StandardCharsets.UTF_8);
    }

    public static Cov loadCoverage(Path p) throws IOException {
        Cov c = new Cov();
        if (!Files.exists(p))
            return c;
        for (String line : Files.readAllLines(p, StandardCharsets.US_ASCII)) {
            line = line.strip();
            if (line.isEmpty() || line.startsWith("#")) continue;
            String[] t = line.split("\\s+");
            if (t[0].equals("x") && t.length == 3) c.exec.add(new String[] { t[1], t[2] });
            else if (t[0].equals("e") && t.length == 5) c.edges.add(new String[] { t[1], t[2], t[3], t[4] });
            else throw new IllegalArgumentException(p + ": bad coverage line: " + line);
        }
        return c;
    }

    /** Address of a record or coverage entry. "ram" coverage (RAM outside RAMCODE) has no block and maps to the default space. */
    public static Address addr(Program p, String space, int a) {
        switch (space) {
            case "ram": case "rom":
                return p.getAddressFactory().getDefaultAddressSpace().getAddress(a);
            case "low": case "ramcode": {
                MemoryBlock b = p.getMemory().getBlock(space.equals("low") ? "LOW" : "RAMCODE");
                if (b == null) throw new IllegalStateException("no " + space + " block: run BuildProject first");
                return b.getStart().getAddressSpace().getAddress(a);
            }
            default:
                throw new IllegalArgumentException("unknown space " + space);
        }
    }

    public static String spaceName(Address a) {
        if (a.getAddressSpace().isOverlaySpace()) {
            String n = a.getAddressSpace().getName();
            if (n.startsWith("LOW")) return "low";
            if (n.startsWith("RAMCODE")) return "ramcode";
        }
        return a.getOffset() < 0x4000 ? "ram" : "rom";
    }
}
```

```java
// Rebuilds the program from the ROM image and ghidra740/annotations (M3 spec §5.2-§5.4).
// Run on a freshly imported UK_V2.01_4.04_ROM.BIN (raw binary, MELPS740:LE:16:default, no analysis).
// Writes build/ghidra740/build.txt: ERROR lines and a SUMMARY line. CheckProject reports them as criterion 0.
//@category ghidra740
import ghidra.app.script.GhidraScript;
import ghidra.program.database.mem.FileBytes;
import ghidra.program.model.address.*;
import ghidra.program.model.data.*;
import ghidra.program.model.listing.*;
import ghidra.program.model.mem.*;
import ghidra.program.model.symbol.*;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.*;

public class BuildProject extends GhidraScript {
    private final List<String> errors = new ArrayList<>();
    private final Set<Address> functionTargets = new TreeSet<>();

    @Override
    public void run() throws Exception {
        Path root = Annot.repoRoot(this);
        Path report = root.resolve("build/ghidra740/build.txt");
        Files.createDirectories(report.getParent());
        String summary = "not built";
        try {
            Annot.Annotations an = Annot.load(root.resolve("ghidra740/annotations"));
            Annot.Cov cov = Annot.loadCoverage(root.resolve("ghidra740/coverage/corpus.cov"));
            buildMemory();
            Map<String, DataType> structs = createStructs(an);
            applyRecords(an, structs);
            List<Address> seeds = new ArrayList<>();
            vectorSeeds(seeds);
            recordSeeds(an, seeds);
            coverageSeeds(cov, seeds);
            for (Address a : seeds)
                if (getInstructionAt(a) == null && !disassemble(a) && getInstructionAt(a) == null)
                    errors.add("cannot disassemble seed " + Annot.spaceName(a) + " " + hex(a));
            coverageEdges(cov);
            createFunctions(an);
            summary = String.format("SUMMARY %d records, %d seeds, %d functions, %d instructions",
                an.recs.size(), seeds.size(), currentProgram.getFunctionManager().getFunctionCount(),
                currentProgram.getListing().getNumInstructions());
        } catch (Exception e) {
            errors.add(e.getClass().getSimpleName() + ": " + e.getMessage());
        }
        StringBuilder sb = new StringBuilder();
        for (String e : errors) sb.append("ERROR ").append(e).append('\n');
        sb.append(summary).append('\n');
        Files.writeString(report, sb.toString(), StandardCharsets.UTF_8);
        println("BuildProject: " + errors.size() + " error(s); " + summary);
        for (String e : errors) printerr("BuildProject: " + e);
    }

    private static String hex(Address a) { return String.format("%04X", a.getOffset()); }

    private void buildMemory() throws Exception {
        Memory mem = currentProgram.getMemory();
        if (mem.getBlock("ROM") != null)
            throw new IllegalStateException("already built: delete the program and import the ROM again");
        List<FileBytes> fbs = mem.getAllFileBytes();
        if (fbs.size() != 1 || fbs.get(0).getSize() != 0x10000)
            throw new IllegalStateException("expected one 64 KB imported file (UK_V2.01_4.04_ROM.BIN)");
        FileBytes fb = fbs.get(0);
        for (MemoryBlock b : mem.getBlocks())
            mem.removeBlock(b, monitor);
        AddressSpace ds = currentProgram.getAddressFactory().getDefaultAddressSpace();
        mem.createUninitializedBlock("ZP", ds.getAddress(0x0000), 0xDA, false).setPermissions(true, true, false);
        MemoryBlock sfr = mem.createUninitializedBlock("SFR", ds.getAddress(0x00DA), 0x26, false);
        sfr.setPermissions(true, true, false);
        sfr.setVolatile(true);
        mem.createUninitializedBlock("RAM", ds.getAddress(0x0100), 0x3F00, false).setPermissions(true, true, true);
        mem.createInitializedBlock("ROM_LO", ds.getAddress(0x4000), fb, 0x4000, 0x4000, false).setPermissions(true, false, true);
        mem.createInitializedBlock("ROM", ds.getAddress(0x8000), fb, 0x8000, 0x8000, false).setPermissions(true, false, true);
        mem.createInitializedBlock("LOW", ds.getAddress(0x4000), fb, 0x0000, 0x4000, true).setPermissions(true, false, true);
        mem.createInitializedBlock("RAMCODE", ds.getAddress(0x0200), fb, 0xADFE, 0x1FF, true).setPermissions(true, true, true);
    }

    private DataType primitive(String base) {
        DataTypeManager dtm = currentProgram.getDataTypeManager();
        switch (base) {
            case "byte": return ByteDataType.dataType;
            case "word": case "ptrm1": return WordDataType.dataType;
            case "ptr": return new PointerDataType(VoidDataType.dataType, 2, dtm);
            case "char": return CharDataType.dataType;
            case "string": return StringDataType.dataType;
            default: return null;
        }
    }

    private DataType dataType(Annot.TypeRef t, Map<String, DataType> structs) {
        DataType e = structs.containsKey(t.base) ? structs.get(t.base) : primitive(t.base);
        if (e == null || !t.array || t.base.equals("string"))
            return e;
        return new ArrayDataType(e, t.count, e.getLength(), currentProgram.getDataTypeManager());
    }

    private Map<String, DataType> createStructs(Annot.Annotations an) {
        DataTypeManager dtm = currentProgram.getDataTypeManager();
        Map<String, DataType> out = new HashMap<>();
        for (Annot.Struct s : an.structs.values()) {
            StructureDataType st = new StructureDataType(new CategoryPath("/" + Annot.BM_CATEGORY), s.name, s.size, dtm);
            if (!s.comment.isEmpty()) st.setDescription(s.comment);
            for (Annot.Field f : s.fields) {
                DataType dt = dataType(Annot.parseType(f.type), Map.of());
                st.replaceAtOffset(f.offset, dt, dt.getLength(), f.name, f.comment.isEmpty() ? null : f.comment);
            }
            out.put(s.name, dtm.addDataType(st, DataTypeConflictHandler.REPLACE_HANDLER));
        }
        return out;
    }

    private void applyRecords(Annot.Annotations an, Map<String, DataType> structs) {
        BookmarkManager bm = currentProgram.getBookmarkManager();
        for (Annot.Rec r : an.recs) {
            try {
                Address a = Annot.addr(currentProgram, r.space, r.addr);
                switch (r.kind) {
                    case "data":
                    case "jumptable": {
                        Annot.TypeRef t = Annot.parseType(r.type);
                        int len = t.base.equals("string") ? t.count : -1;
                        DataUtilities.createData(currentProgram, a, dataType(t, structs), len,
                            DataUtilities.ClearDataMode.CHECK_FOR_SPACE);
                        createLabel(a, r.name, true, SourceType.USER_DEFINED);
                        if (!r.comment.isEmpty()) setEOLComment(a, r.comment);
                        if (r.kind.equals("jumptable")) {
                            bm.setBookmark(a, BookmarkType.NOTE, Annot.BM_CATEGORY, t.base.equals("ptrm1") ? "jumptable-m1" : "jumptable");
                            jumptableTargets(a, t);
                        }
                        break;
                    }
                    case "label":
                        createLabel(a, r.name, true, SourceType.USER_DEFINED);
                        if (!r.comment.isEmpty()) setEOLComment(a, r.comment);
                        break;
                    case "comment":
                        setEOLComment(a, r.comment);
                        break;
                    case "entry":
                        bm.setBookmark(a, BookmarkType.NOTE, Annot.BM_CATEGORY, r.comment.isEmpty() ? "entry" : "entry: " + r.comment);
                        break;
                    default:   // func: created after disassembly
                        break;
                }
            } catch (Exception e) {
                errors.add(r.where() + ": " + e.getMessage());
            }
        }
    }

    /** Targets of a code-pointer table, in the default space. ptrm1 entries hold target - 1. */
    private void jumptableTargets(Address table, Annot.TypeRef t) throws Exception {
        AddressSpace ds = currentProgram.getAddressFactory().getDefaultAddressSpace();
        for (int i = 0; i < t.count; i++) {
            int v = currentProgram.getMemory().getShort(table.add(2L * i)) & 0xFFFF;
            if (t.base.equals("ptrm1")) {
                v = (v + 1) & 0xFFFF;
                currentProgram.getReferenceManager().addMemoryReference(table.add(2L * i), ds.getAddress(v),
                    RefType.DATA, SourceType.USER_DEFINED, 0);
            }
            functionTargets.add(ds.getAddress(v));
        }
    }

    private void vectorSeeds(List<Address> seeds) throws Exception {
        AddressSpace ds = currentProgram.getAddressFactory().getDefaultAddressSpace();
        for (int v = 0xFFF4; v <= 0xFFFE; v += 2) {
            Address t = ds.getAddress(currentProgram.getMemory().getShort(ds.getAddress(v)) & 0xFFFF);
            seeds.add(t);
            functionTargets.add(t);
        }
    }

    private void recordSeeds(Annot.Annotations an, List<Address> seeds) {
        for (Annot.Rec r : an.recs)
            if (r.kind.equals("func") || r.kind.equals("entry"))
                seeds.add(Annot.addr(currentProgram, r.space, r.addr));
        seeds.addAll(functionTargets);   // jumptable targets collected by applyRecords
    }

    private void coverageSeeds(Annot.Cov cov, List<Address> seeds) {
        for (String[] x : cov.exec) {
            if (x[0].equals("ram")) {
                errors.add("coverage: execution in RAM outside RAMCODE at " + x[1] + " (not disassembled)");
                continue;
            }
            seeds.add(Annot.addr(currentProgram, x[0], Integer.parseInt(x[1], 16)));
        }
    }

    private void coverageEdges(Annot.Cov cov) {
        ReferenceManager rm = currentProgram.getReferenceManager();
        for (String[] e : cov.edges) {
            Address from = Annot.addr(currentProgram, e[0], Integer.parseInt(e[1], 16));
            Address to = Annot.addr(currentProgram, e[2], Integer.parseInt(e[3], 16));
            Instruction in = getInstructionAt(from);
            if (in == null) {
                errors.add("coverage edge from " + e[0] + " " + e[1] + ": no instruction");
                continue;
            }
            try {
                int op = in.getByte(0) & 0xFF;
                rm.addMemoryReference(from, to, op == 0x02 ? RefType.COMPUTED_CALL : RefType.COMPUTED_JUMP,
                    SourceType.USER_DEFINED, Reference.MNEMONIC);
                functionTargets.add(to);
            } catch (Exception ex) {
                errors.add("coverage edge " + String.join(" ", e) + ": " + ex.getMessage());
            }
        }
    }

    private void createFunctions(Annot.Annotations an) throws Exception {
        for (Annot.Rec r : an.recs) {
            if (!r.kind.equals("func")) continue;
            Address a = Annot.addr(currentProgram, r.space, r.addr);
            if (getInstructionAt(a) == null) { errors.add(r.where() + ": no instruction at function entry"); continue; }
            Function f = getFunctionAt(a);
            if (f == null) f = createFunction(a, r.name);
            else f.setName(r.name, SourceType.USER_DEFINED);
            if (f == null) { errors.add(r.where() + ": cannot create function " + r.name); continue; }
            if (!r.comment.isEmpty()) setPlateComment(a, r.comment);
        }
        InstructionIterator it = currentProgram.getListing().getInstructions(true);
        while (it.hasNext()) {
            Instruction in = it.next();
            if (in.getFlowType().isCall())
                for (Address t : in.getFlows())
                    functionTargets.add(t);
        }
        for (Address t : functionTargets)
            if (getFunctionAt(t) == null && getInstructionAt(t) != null && createFunction(t, null) == null)
                errors.add("cannot create a function at " + Annot.spaceName(t) + " " + hex(t));
    }
}
```

```java
// Checks the project against the M3 done criteria (M3 spec §6) and writes build/ghidra740/check.txt.
//@category ghidra740
import ghidra.app.script.GhidraScript;
import ghidra.program.model.address.*;
import ghidra.program.model.listing.*;
import ghidra.program.model.mem.*;
import ghidra.program.model.symbol.*;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.*;

public class CheckProject extends GhidraScript {
    private static final List<String> CODE_BLOCKS = List.of("ROM_LO", "ROM", "LOW", "RAMCODE");
    private static final Set<String> NEVER_USED = Set.of("SET", "SED", "MUL", "DIV", "STP", "BRK");
    private final StringBuilder out = new StringBuilder();
    private boolean pass = true;

    @Override
    public void run() throws Exception {
        Path root = Annot.repoRoot(this);
        Path report = root.resolve("build/ghidra740/check.txt");
        Files.createDirectories(report.getParent());
        out.append("INFO 1 module: checked separately by ghidra740\\validate.bat\n");
        if (checkBuild(root)) {
            AddressSet code = new AddressSet();
            for (String b : CODE_BLOCKS) {
                MemoryBlock blk = currentProgram.getMemory().getBlock(b);
                code.add(blk.getStart(), blk.getEnd());
            }
            dynamicCoverage(Annot.loadCoverage(root.resolve("ghidra740/coverage/corpus.cov")));
            staticCoverage(code);
            indirectFlow(code);
            names(code);
            census(code);
        }
        out.append(pass ? "RESULT PASS\n" : "RESULT FAIL\n");
        Files.writeString(report, out.toString(), StandardCharsets.UTF_8);
        for (String line : out.toString().split("\n"))
            if (!line.startsWith("  "))
                println(line);
    }

    private void report(int n, String name, String detail, List<String> bad) {
        if (bad.isEmpty()) {
            out.append(String.format("PASS %d %s: %s%n", n, name, detail));
            return;
        }
        pass = false;
        out.append(String.format("FAIL %d %s: %s, %d problem(s)%n", n, name, detail, bad.size()));
        for (String b : bad) out.append("  ").append(b).append('\n');
    }

    private String where(Address a) { return Annot.spaceName(a) + " " + String.format("%04X", a.getOffset()); }

    private boolean checkBuild(Path root) throws Exception {
        List<String> bad = new ArrayList<>();
        Path build = root.resolve("build/ghidra740/build.txt");
        if (!Files.exists(build)) bad.add("no build/ghidra740/build.txt: BuildProject did not run");
        else for (String l : Files.readAllLines(build, StandardCharsets.UTF_8)) if (l.startsWith("ERROR ")) bad.add(l.substring(6));
        for (String b : CODE_BLOCKS)
            if (currentProgram.getMemory().getBlock(b) == null) bad.add("missing memory block " + b);
        report(0, "build", "BuildProject report and memory blocks", bad);
        return CODE_BLOCKS.stream().allMatch(b -> currentProgram.getMemory().getBlock(b) != null);
    }

    private void dynamicCoverage(Annot.Cov cov) {
        List<String> bad = new ArrayList<>();
        int n = 0;
        for (String[] x : cov.exec) {
            if (x[0].equals("ram")) { bad.add("ram " + x[1] + ": executed in RAM outside RAMCODE"); continue; }
            Address a = Annot.addr(currentProgram, x[0], Integer.parseInt(x[1], 16));
            n++;
            if (getInstructionAt(a) == null) bad.add(where(a) + ": not an instruction start");
            else if (getFunctionContaining(a) == null) bad.add(where(a) + ": not inside a function");
        }
        report(2, "dynamic-coverage", n + " executed addresses", bad);
    }

    private void staticCoverage(AddressSet code) throws Exception {
        List<String> bad = new ArrayList<>();
        long undefined = 0;
        for (AddressRange r : currentProgram.getListing().getUndefinedRanges(code, true, monitor)) {
            undefined += r.getLength();
            bad.add(where(r.getMinAddress()) + "-" + String.format("%04X", r.getMaxAddress().getOffset()) + ": undefined (" + r.getLength() + " bytes)");
        }
        Iterator<Bookmark> it = currentProgram.getBookmarkManager().getBookmarksIterator(BookmarkType.ERROR);
        while (it.hasNext()) {
            Bookmark b = it.next();
            bad.add(where(b.getAddress()) + ": " + b.getCategory() + ": " + b.getComment());
        }
        report(3, "static-coverage", undefined + " undefined bytes in " + String.join("/", CODE_BLOCKS), bad);
    }

    private static boolean isIndirect(int op) { return op == 0x6C || op == 0xB2 || op == 0x02; }

    private void indirectFlow(AddressSet code) throws Exception {
        List<String> bad = new ArrayList<>();
        int n = 0;
        InstructionIterator it = currentProgram.getListing().getInstructions(code, true);
        while (it.hasNext()) {
            Instruction in = it.next();
            if (!isIndirect(in.getByte(0) & 0xFF)) continue;
            n++;
            List<Address> targets = new ArrayList<>();
            for (Reference r : in.getReferencesFrom())
                if (r.getReferenceType().isComputed()) targets.add(r.getToAddress());
            if (targets.isEmpty() && getEOLComment(in.getAddress()) == null)
                bad.add(where(in.getAddress()) + " " + in + ": no resolved targets and no comment");
            for (Address t : targets)
                if (getFunctionAt(t) == null) bad.add(where(in.getAddress()) + " " + in + ": target " + where(t) + " is not a function entry");
        }
        report(4, "indirect-flow", n + " indirect JMP/JSR", bad);
    }

    private boolean named(Address a) {
        Symbol s = getSymbolAt(a);
        if (s != null && s.getSource() != SourceType.DEFAULT) return true;
        Data d = currentProgram.getListing().getDataContaining(a);
        Symbol ds = d == null ? null : d.getPrimarySymbol();
        return ds != null && ds.getSource() != SourceType.DEFAULT;
    }

    /** RAM base address used by (zp,X), (zp),Y, JMP/JSR (zp) or abs,X / abs,Y below $4000; null otherwise. */
    private static Integer ramBase(Instruction in) throws Exception {
        int op = in.getByte(0) & 0xFF;
        boolean zpPtr = (op & 0x1F) == 0x01 || (op & 0x1F) == 0x11 || op == 0x02 || op == 0xB2;
        boolean absIdx = (op & 0x1F) == 0x1D || (op & 0x1F) == 0x19 || op == 0x1E || op == 0x3E || op == 0x5E
            || op == 0x7E || op == 0xDE || op == 0xFE || op == 0xBC || op == 0xBE;
        if (zpPtr) return in.getByte(1) & 0xFF;
        if (absIdx) {
            int a = (in.getByte(1) & 0xFF) | (in.getByte(2) & 0xFF) << 8;
            return a < 0x4000 ? a : null;
        }
        return null;
    }

    private void names(AddressSet code) throws Exception {
        List<String> bad = new ArrayList<>();
        int functions = 0, data = 0;
        for (Function f : currentProgram.getListing().getFunctions(true)) {
            functions++;
            if (f.getSymbol().getSource() == SourceType.DEFAULT) bad.add(where(f.getEntryPoint()) + ": default function name " + f.getName());
            if (getPlateComment(f.getEntryPoint()) == null) bad.add(where(f.getEntryPoint()) + ": " + f.getName() + " has no plate comment");
        }
        DataIterator di = currentProgram.getListing().getDefinedData(code, true);
        while (di.hasNext()) {
            Data d = di.next();
            data++;
            Symbol s = d.getPrimarySymbol();
            if (s == null || s.getSource() == SourceType.DEFAULT) bad.add(where(d.getAddress()) + ": unnamed data " + d.getDataType().getName());
        }
        Set<Integer> bases = new TreeSet<>();
        InstructionIterator it = currentProgram.getListing().getInstructions(code, true);
        while (it.hasNext()) {
            Integer b = ramBase(it.next());
            if (b != null) bases.add(b);
        }
        AddressSpace ds = currentProgram.getAddressFactory().getDefaultAddressSpace();
        for (int b : bases)
            if (!named(ds.getAddress(b))) bad.add(String.format("ram %04X: RAM pointer/buffer base has no name", b));
        report(5, "names", functions + " functions, " + data + " data items, " + bases.size() + " RAM bases", bad);
    }

    private void census(AddressSet code) {
        Map<String, Integer> counts = new TreeMap<>();
        Map<String, List<String>> where = new TreeMap<>();
        InstructionIterator it = currentProgram.getListing().getInstructions(code, true);
        while (it.hasNext()) {
            Instruction in = it.next();
            String m = in.getMnemonicString().toUpperCase(Locale.ROOT).replaceAll("[0-7]$", "");
            counts.merge(m, 1, Integer::sum);
            if (NEVER_USED.contains(m)) where.computeIfAbsent(m, k -> new ArrayList<>()).add(where(in.getAddress()));
        }
        for (Map.Entry<String, Integer> e : counts.entrySet()) out.append(String.format("CENSUS %-4s %d%n", e.getKey(), e.getValue()));
        for (Map.Entry<String, List<String>> e : where.entrySet())
            out.append("FINDING 6 ").append(e.getKey()).append(" used at ").append(String.join(", ", e.getValue())).append('\n');
        report(6, "opcode-census", counts.size() + " mnemonics" + (where.isEmpty() ? ", none of SET/SED/MUL/DIV/STP/BRK" : ", see FINDING lines"), List.of());
    }
}
```

```java
// Writes the program's names, types, comments and seeds back to the annotation files (M3 spec §5.3).
// Args: "write" (default) rewrites ghidra740/annotations; "check" writes build/ghidra740/export and compares
// (build/ghidra740/roundtrip.txt). A record keeps the file it came from; new records go to new.txt.
//@category ghidra740
import ghidra.app.script.GhidraScript;
import ghidra.program.model.address.*;
import ghidra.program.model.data.*;
import ghidra.program.model.data.Array;
import ghidra.program.model.listing.*;
import ghidra.program.model.symbol.*;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.*;
import java.util.stream.*;

public class ExportAnnotations extends GhidraScript {
    private final Map<String, String> fileOfKey = new HashMap<>(), fileOfAddr = new HashMap<>();
    private final Set<String> named = new HashSet<>(), eolUsed = new HashSet<>();
    private final Map<String, String> tableKind = new HashMap<>();   // addr key -> jumptable | jumptable-m1

    @Override
    public void run() throws Exception {
        String mode = getScriptArgs().length > 0 ? getScriptArgs()[0] : "write";
        Path root = Annot.repoRoot(this);
        Path dir = root.resolve("ghidra740/annotations");
        Annot.Annotations old = Annot.load(dir);
        for (Annot.Rec r : old.recs) {
            fileOfKey.put(key(r.space, r.addr) + ":" + r.kind, r.file);
            fileOfAddr.putIfAbsent(key(r.space, r.addr), r.file);
        }
        Annot.Annotations now = new Annot.Annotations();
        now.headers.putAll(old.headers);
        collect(now);
        collectStructs(now);
        if (mode.equals("write")) {
            Annot.write(now, dir);
            println("ExportAnnotations: wrote " + now.recs.size() + " records to " + dir);
            return;
        }
        Path tmp = root.resolve("build/ghidra740/export");
        Files.createDirectories(tmp);
        try (Stream<Path> s = Files.list(tmp)) {
            for (Path p : s.collect(Collectors.toList())) Files.delete(p);
        }
        Annot.write(now, tmp);
        Set<String> files = new TreeSet<>();
        try (Stream<Path> a = Files.list(dir); Stream<Path> b = Files.list(tmp)) {
            a.forEach(p -> files.add(p.getFileName().toString()));
            b.forEach(p -> files.add(p.getFileName().toString()));
        }
        StringBuilder sb = new StringBuilder();
        for (String f : files) {
            Path x = dir.resolve(f), y = tmp.resolve(f);
            if (!Files.exists(x) || !Files.exists(y) || !Files.readString(x).equals(Files.readString(y)))
                sb.append("DIFF ").append(f).append('\n');
        }
        sb.append(sb.length() == 0 ? "RESULT PASS\n" : "RESULT FAIL\n");
        Files.writeString(root.resolve("build/ghidra740/roundtrip.txt"), sb.toString(), StandardCharsets.UTF_8);
        println("ExportAnnotations check: " + sb.toString().replace('\n', ' '));
    }

    private static String key(String space, long addr) { return space + ":" + String.format("%04X", addr); }
    private static String key(Address a) { return key(Annot.spaceName(a), a.getOffset()); }
    private static String oneLine(String s) { return s == null ? "" : s.replaceAll("\\s*[\\r\\n]+\\s*", " ").strip(); }

    private void add(Annot.Annotations now, Address a, String kind, String name, String type, String comment) {
        String k = key(a);
        String file = fileOfKey.getOrDefault(k + ":" + kind, fileOfAddr.getOrDefault(k, Annot.NEW_FILE));
        now.recs.add(new Annot.Rec(file, 0, Annot.spaceName(a), (int) a.getOffset(), kind, name, type, oneLine(comment)));
    }

    private String eol(Address a) {
        String c = getEOLComment(a);
        if (c != null) eolUsed.add(key(a));
        return c;
    }

    private void collect(Annot.Annotations now) throws Exception {
        Listing listing = currentProgram.getListing();
        Iterator<Bookmark> bit = currentProgram.getBookmarkManager().getBookmarksIterator(BookmarkType.NOTE);
        while (bit.hasNext()) {
            Bookmark b = bit.next();
            if (!b.getCategory().equals(Annot.BM_CATEGORY)) continue;
            String d = b.getComment();
            if (d.equals("jumptable") || d.equals("jumptable-m1")) tableKind.put(key(b.getAddress()), d);
            else if (d.equals("entry")) add(now, b.getAddress(), "entry", "-", "-", "");
            else if (d.startsWith("entry: ")) add(now, b.getAddress(), "entry", "-", "-", d.substring(7));
        }
        for (Function f : listing.getFunctions(true)) {
            if (f.getSymbol().getSource() == SourceType.DEFAULT) continue;
            Address a = f.getEntryPoint();
            add(now, a, "func", f.getName(), "-", getPlateComment(a));
            named.add(key(a));
        }
        DataIterator di = listing.getDefinedData(true);
        while (di.hasNext()) {
            Data d = di.next();
            Symbol s = d.getPrimarySymbol();
            if (s == null || s.getSource() == SourceType.DEFAULT) continue;
            Address a = d.getAddress();
            String tk = tableKind.get(key(a));
            add(now, a, tk == null ? "data" : "jumptable", s.getName(), typeString(d, "jumptable-m1".equals(tk)), eol(a));
            named.add(key(a));
        }
        SymbolIterator si = currentProgram.getSymbolTable().getAllSymbols(false);
        while (si.hasNext()) {
            Symbol s = si.next();
            if (s.getSymbolType() != SymbolType.LABEL || s.getSource() == SourceType.DEFAULT || !s.isPrimary()) continue;
            Address a = s.getAddress();
            if (!a.isMemoryAddress() || named.contains(key(a))) continue;
            add(now, a, "label", s.getName(), "-", eol(a));
            named.add(key(a));
        }
        AddressIterator ci = listing.getCommentAddressIterator(CommentType.EOL, currentProgram.getMemory(), true);
        while (ci.hasNext()) {
            Address a = ci.next();
            if (eolUsed.contains(key(a))) continue;
            add(now, a, "comment", "-", "-", getEOLComment(a));
        }
    }

    private static String base(DataType dt, boolean m1) {
        if (dt instanceof Pointer) return "ptr";
        if (dt instanceof ByteDataType) return "byte";
        if (dt instanceof WordDataType) return m1 ? "ptrm1" : "word";
        if (dt instanceof CharDataType) return "char";
        if (dt instanceof Structure) return dt.getName();
        return "?" + dt.getName();   // fails validation on the next load, so it gets noticed
    }

    private static String typeString(Data d, boolean m1) {
        DataType dt = d.getDataType();
        if (dt instanceof Array) {
            Array arr = (Array) dt;
            return base(arr.getDataType(), m1) + "[" + arr.getNumElements() + "]";
        }
        if (dt instanceof StringDataType) return "string[" + d.getLength() + "]";
        return base(dt, m1);
    }

    private void collectStructs(Annot.Annotations now) {
        Category cat = currentProgram.getDataTypeManager().getCategory(new CategoryPath("/" + Annot.BM_CATEGORY));
        if (cat == null) return;
        for (DataType dt : cat.getDataTypes()) {
            if (!(dt instanceof Structure)) continue;
            Structure st = (Structure) dt;
            Annot.Struct s = new Annot.Struct(st.getName(), st.getLength(), oneLine(st.getDescription()));
            for (DataTypeComponent c : st.getDefinedComponents()) {
                DataType ct = c.getDataType();
                String type = ct instanceof Array
                    ? base(((Array) ct).getDataType(), false) + "[" + ((Array) ct).getNumElements() + "]"
                    : base(ct, false);
                s.fields.add(new Annot.Field(c.getOffset(), c.getFieldName(), type, oneLine(c.getComment())));
            }
            now.structs.put(s.name, s);
        }
    }
}
```

```java
// Writes per-function disassembly and decompiled C plus indexes to build/ghidra740/dump/ (M3 spec §7),
// for the labelling loop. Read-only.
//@category ghidra740
import ghidra.app.decompiler.*;
import ghidra.app.script.GhidraScript;
import ghidra.program.model.address.*;
import ghidra.program.model.listing.*;
import ghidra.program.model.mem.*;
import ghidra.program.model.symbol.*;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.util.*;
import java.util.stream.*;

public class DumpProject extends GhidraScript {
    @Override
    public void run() throws Exception {
        Path out = Annot.repoRoot(this).resolve("build/ghidra740/dump");
        Files.createDirectories(out);
        try (Stream<Path> s = Files.list(out)) {
            for (Path p : s.collect(Collectors.toList())) Files.delete(p);
        }
        Listing listing = currentProgram.getListing();
        CodeUnitFormat fmt = new CodeUnitFormat(new CodeUnitFormatOptions());
        DecompInterface di = new DecompInterface();
        di.openProgram(currentProgram);
        StringBuilder index = new StringBuilder("# space addr  size name | called by | calls\n");
        for (Function f : listing.getFunctions(true)) {
            monitor.checkCancelled();
            Address e = f.getEntryPoint();
            String tag = Annot.spaceName(e) + "_" + String.format("%04X", e.getOffset());
            String callers = names(f.getCallingFunctions(monitor)), callees = names(f.getCalledFunctions(monitor));
            index.append(String.format("%-7s %04X %5d %s | %s | %s%n", Annot.spaceName(e), e.getOffset(),
                f.getBody().getNumAddresses(), f.getName(), callers, callees));
            StringBuilder sb = new StringBuilder();
            sb.append("// ").append(f.getName()).append(" @ ").append(tag).append('\n');
            String plate = getPlateComment(e);
            if (plate != null) sb.append("// ").append(plate.replace("\n", "\n// ")).append('\n');
            sb.append("// called by: ").append(callers).append("\n// calls: ").append(callees).append("\n\n");
            for (CodeUnit cu : listing.getCodeUnits(f.getBody(), true)) {
                for (Symbol s : currentProgram.getSymbolTable().getSymbols(cu.getAddress()))
                    sb.append(s.getName()).append(":\n");
                String eol = cu.getComment(CommentType.EOL);
                sb.append(String.format("  %04X  %-9s %s%s%n", cu.getAddress().getOffset(), bytes(cu),
                    fmt.getRepresentationString(cu), eol == null ? "" : "    ; " + eol));
            }
            DecompileResults r = di.decompileFunction(f, 60, monitor);
            sb.append("\n/* decompiled */\n").append(r.decompileCompleted() && r.getDecompiledFunction() != null
                ? r.getDecompiledFunction().getC() : "/* decompile failed: " + r.getErrorMessage() + " */\n");
            Files.writeString(out.resolve(tag + "_" + f.getName() + ".txt"), sb.toString(), StandardCharsets.UTF_8);
        }
        di.dispose();
        Files.writeString(out.resolve("functions.txt"), index.toString(), StandardCharsets.UTF_8);

        StringBuilder data = new StringBuilder("# space addr  len type name ; comment | referenced from\n");
        DataIterator dit = listing.getDefinedData(true);
        while (dit.hasNext()) {
            Data d = dit.next();
            String eol = d.getComment(CommentType.EOL);
            data.append(String.format("%-7s %04X %5d %-14s %s%s | %s%n", Annot.spaceName(d.getAddress()), d.getAddress().getOffset(),
                d.getLength(), d.getDataType().getName(), d.getLabel() == null ? "-" : d.getLabel(),
                eol == null ? "" : " ; " + eol, refs(d.getAddress())));
        }
        Files.writeString(out.resolve("data.txt"), data.toString(), StandardCharsets.UTF_8);

        AddressSet code = new AddressSet();
        for (String b : List.of("ROM_LO", "ROM", "LOW", "RAMCODE")) {
            MemoryBlock blk = currentProgram.getMemory().getBlock(b);
            code.add(blk.getStart(), blk.getEnd());
        }
        StringBuilder undef = new StringBuilder("# space start-end len | first bytes | referenced from (first address)\n");
        for (AddressRange r : listing.getUndefinedRanges(code, true, monitor)) {
            byte[] b = new byte[(int) Math.min(16, r.getLength())];
            currentProgram.getMemory().getBytes(r.getMinAddress(), b);
            StringBuilder hx = new StringBuilder();
            for (byte x : b) hx.append(String.format("%02X ", x & 0xFF));
            undef.append(String.format("%-7s %04X-%04X %5d | %s| %s%n", Annot.spaceName(r.getMinAddress()), r.getMinAddress().getOffset(),
                r.getMaxAddress().getOffset(), r.getLength(), hx, refs(r.getMinAddress())));
        }
        Files.writeString(out.resolve("undefined.txt"), undef.toString(), StandardCharsets.UTF_8);
        println("DumpProject: wrote " + out);
    }

    private static String names(Set<Function> fs) {
        return fs.stream().map(Function::getName).sorted().collect(Collectors.joining(" "));
    }

    private static String bytes(CodeUnit cu) throws Exception {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < Math.min(cu.getLength(), 3); i++) sb.append(String.format("%02X", cu.getByte(i) & 0xFF));
        return sb.toString();
    }

    private String refs(Address a) {
        List<String> r = new ArrayList<>();
        for (Reference ref : currentProgram.getReferenceManager().getReferencesTo(a)) {
            if (r.size() == 8) { r.add("..."); break; }
            r.add(Annot.spaceName(ref.getFromAddress()) + ":" + String.format("%04X", ref.getFromAddress().getOffset()));
        }
        return String.join(" ", r);
    }
}
```
